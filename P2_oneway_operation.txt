==================== oneway.java (package tyc9) ====================
package tyc9;

import java.sql.DriverManager;
import javax.jws.WebService;
import javax.jws.WebMethod;
import javax.jws.WebParam;

@WebService(serviceName = "oneway")
public class oneway {

    @WebMethod(operationName = "hello")
    public String hello(@WebParam(name = "name") String txt) {
        return "Hello " + txt + " !";
    }

    @WebMethod(operationName = "insertstudent")
    public String insertstudent(
            @WebParam(name = "id") int id,
            @WebParam(name = "name") String name,
            @WebParam(name = "sclass") final String sclass) {
        try {
            Class.forName("com.mysql.jdbc.Driver");
            java.sql.Connection con = DriverManager.getConnection(
                    "jdbc:mysql://localhost:3306/ty49", "root", "mysql");
            java.sql.PreparedStatement ps = con.prepareStatement(
                    "INSERT INTO student VALUES (?, ?, ?)");
            ps.setInt(1, id);
            ps.setString(2, name);
            ps.setString(3, sclass);
            int i = ps.executeUpdate();
            ps.close();
            con.close();
            return i + " record inserted";
        } catch (Exception e) {
            return "Error = " + e;
        }
    }
}
