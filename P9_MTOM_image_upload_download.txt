MTOM (Message Transmission Optimization Mechanism) web service: image upload / download

What is MTOM?
MTOM allows web services to send binary content, such as images or files,
without converting it to base64 text within the SOAP envelope. Normally,
binary data is converted to xs:base64Binary in XML, which increases the data
size by approximately 33% and adds processing overhead. MTOM overcomes this
by transmitting binary data as separate MIME attachments while keeping it
logically part of the SOAP message.

package tycs;
import java.io.*;
import javax.jws.Oneway;
import javax.jws.WebService;
import javax.jws.WebMethod;
import javax.jws.WebParam;
import javax.xml.ws.soap.MTOM;
/**
 *
 * @author Administrator
 */
@MTOM(enabled=true, threshold=6000)
@WebService(serviceName = "imagews1")
public class imagews1 {

  /**
   * This is a sample web service operation
   */
  @WebMethod(operationName = "hello")
  public String hello(@WebParam(name = "name") String txt) {
     return "Hello " + txt + " !";
  }

  /**
   * Web service operation
   */
  @WebMethod(operationName = "upload")
  @Oneway
  public void upload(@WebParam(name = "Filename") String Filename, @WebParam(name =
"ImageBytes") byte[] ImageBytes) {
     String filePath = "C:\\Users\\Administrator\\Downloads\\" +Filename;
     try{FileOutputStream fos = new FileOutputStream(filePath);
          BufferedOutputStream bos = new BufferedOutputStream(fos);
          bos.write(ImageBytes);
   System.out.println("Received file:" +filePath);
  } catch (Exception ex) {
  System.out.println(ex);
}
}

    /**
     * Web service operation
     */
    @WebMethod(operationName = "download")
    public byte[] download(@WebParam(name = "Filename") String Filename) {
       String filePath = "C:\\Users\\Administrator\\Downloads\\" +Filename;
        System.out.println("Sending file:" +filePath);
        try{FileInputStream fis = new FileInputStream(new File(filePath));
        BufferedInputStream bis = new BufferedInputStream(fis);
           byte[] fileBytes = new byte[(int) new File(filePath).length()];
           bis.read(fileBytes);
           return fileBytes;
        } catch (Exception ex) {
           System.out.println(ex);
        }
        return null;
        }

    }

  String action = request.getParameter("action");
  String filename = request.getParameter("filename");
  if ("upload".equals(action) && filename != null) {
        java.io.File file = new java.io.File("C:\\Users\\Administrator\\Downloads\\" + filename);

            byte[] bytes = new byte[(int) file.length()];
            java.io.FileInputStream fis = null;
            try {
                fis = new java.io.FileInputStream(file);
                fis.read(bytes);
            } finally {
                if (fis != null) { fis.close(); }
            }
            port.upload(filename, bytes);

           } else {
               out.println("<p style='color:red;'>File not found on server.</p>");
           }
       }
   } catch (Exception e) {
